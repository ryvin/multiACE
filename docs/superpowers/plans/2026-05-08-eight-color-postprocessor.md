# 8-Color Post-Processor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Enable operator-slice-and-forget 8-color prints on the Snapmaker U1 with multiACE, via a post-processor that auto-resolves filament-to-slot mapping, rewrites gcode, writes a sidecar, and surfaces resolution status in a new Print queue web tab.

**Architecture:** Single-file Python post-processor (`tools/multiace_postprocess.py`, stdlib only) runs as Orca's post-script; queries `/api/slots` for Spoolman bindings; auto-matches by exact `(material, color_hex)`; generates swap plan (greedy); rewrites gcode; writes a `<file>.multiace.json` sidecar. A new FastAPI endpoint pair (`/api/print_queue`, `/api/print_queue/{gcode}/revalidate`) aggregates sidecars from Moonraker's gcode file list. A new "Print queue" tab in vanilla JS reads those endpoints and surfaces the resolution table plus "Fix loadout" wizard (delegating to smart-swap from the web-ops spec).

**Tech Stack:** Python 3.11+ stdlib only (post-processor); FastAPI + httpx (backend endpoints); Vanilla JS (frontend tab); pytest + respx (unit tests); Playwright sync API (structural smoke tests).

    index: int

    color: str         # lowercase 6-char hex without '#', e.g. "ff0000"

    spool_id: int
    spool_name: Optional[str]


@dataclass

# Step 1: Parse gcode header

_SEMICOLON_KV = re.compile(r"^;\s*([\w]+)\s*=\s*(.+)$")


def _normalize_color(raw: str) -> str:
    """Strip leading '#', lowercase, pad to 6 chars if needed."""
    s = raw.strip().lstrip("#").lower()
    if len(s) == 3:
        s = "".join(c * 2 for c in s)
    return s[:6].ljust(6, "0")


def parse_header(lines: list[str]) -> Optional[list[dict]]:
    """Parse Snapmaker Orca gcode header lines into per-tool metadata.

    Returns list of {'type': str, 'color': str} dicts (indices 0-N) or
    None if the header lacks filament_type / filament_colour comments or
    fewer than 2 tools are declared.
    """
    kv: dict[str, str] = {}
    for line in lines:
        m = _SEMICOLON_KV.match(line.strip())
        if m:
            kv[m.group(1)] = m.group(2).strip()
        # Stop scanning once we hit actual gcode
        if line.strip() and not line.strip().startswith(";") and not line.strip().startswith("\n"):
            if kv:  # we already collected some header; stop
                break

    types_raw = kv.get("filament_type", "")
    colors_raw = kv.get("filament_colour", "")
    if not types_raw or not colors_raw:
        return None

    types = [t.strip() for t in types_raw.split(";")]
    colors = [_normalize_color(c) for c in colors_raw.split(";")]

    if len(types) < 2 or len(colors) < 2:
        return None

    count = min(len(types), len(colors))
    return [{"type": types[i].upper(), "color": colors[i]} for i in range(count)]

# CLI entry point

def _printer_url() -> str:
    host = (
        os.environ.get("DAVINCI_U1_HOST")
        or os.environ.get("MULTIACE_PRINTER_URL")
        or "192.168.1.136"
    )
    return f"http://{host}"


def main(argv: list[str] = sys.argv) -> int:
    if len(argv) < 2:
        print("Usage: multiace_postprocess.py <gcode_path>", file=sys.stderr)
        return 1

    gcode_path = Path(argv[1])
    if not gcode_path.exists():
        print(f"Error: gcode file not found: {gcode_path}", file=sys.stderr)
        return 1

    _info(f"Processing {gcode_path}")
    lines = gcode_path.read_text(encoding="utf-8", errors="replace").splitlines()

    tools = parse_header(lines)
    if tools is None:
        print("[postprocess] ERROR: no 8-tool slicer header detected. "
              "Did you select the multiACE 8-color profile?", file=sys.stderr)
        # Write error sidecar
        _write_error_sidecar(gcode_path, "no_8_tool_header")
        return 0   # exit 0 so slicer continues; web UI surfaces the error

    _info(f"Found {len(tools)} tools in header")
    printer_url = _printer_url()

    try:
        slots_resp = query_slots(printer_url)
    except Exception as e:
        _warn(f"Could not reach {printer_url}: {e}")
        _write_error_sidecar(gcode_path, "moonraker_unreachable",
                             tools=tools, errors=[str(e)])
        return 0

    resolutions = match_tools(tools, slots_resp)
    swaps = plan_swaps(resolutions, lines)
    new_lines = rewrite_gcode(lines, resolutions, swaps)

    # Determine overall status
    unresolved = [r for r in resolutions if r.match_quality != "exact" and r.resolved is None]
    ambiguous = [r for r in resolutions if r.match_quality == "ambiguous"]
    if ambiguous:
        status, reason = "pending", "ambiguous_match"
    elif unresolved:
        status, reason = "pending", "missing_bindings"
    else:
        status, reason = "ready", None

    # Write rewritten gcode in place
    gcode_path.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
    write_sidecar(gcode_path, resolutions, swaps, status, reason)
    _info(f"Done. status={status}")
    return 0


def _write_error_sidecar(gcode_path: Path, reason: str,
                          tools: Optional[list] = None,
                          errors: Optional[list] = None) -> None:
    sidecar = {
        "schema": 1,
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "gcode_path": str(gcode_path),
        "status": "error" if reason == "no_8_tool_header" else "pending",
        "reason": reason,
        "tools": {},
        "swaps": [],
        "errors": errors or [],
    }
    if tools:
        for i, t in enumerate(tools):
            sidecar["tools"][str(i)] = {
                "type": t["type"], "color": t["color"],
                "match_quality": "none", "candidates": [],
                "resolved": None, "physical_head": None,
            }
    _atomic_write_json(gcode_path.with_suffix(gcode_path.suffix + ".multiace.json"), sidecar)


def _atomic_write_json(path: Path, data: dict) -> None:
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(data, indent=2), encoding="utf-8")
    tmp.replace(path)


if __name__ == "__main__":
    sys.exit(main())

def query_slots(printer_url: str) -> dict:
    """GET /multiace/api/slots and return the parsed JSON dict.

    Uses urllib.request (stdlib only). Raises on network error or non-200.
    """
    url = f"{printer_url.rstrip('/')}/multiace/api/slots"
    _dbg(f"GET {url}")
    try:
        with urllib.request.urlopen(url, timeout=10) as resp:
            return json.loads(resp.read().decode("utf-8"))
    except urllib.error.HTTPError as e:
        raise RuntimeError(f"GET {url} failed: HTTP {e.code}") from e
    except Exception as e:
        raise RuntimeError(f"GET {url} failed: {e}") from e


def match_tools(tools: list[dict], slots_response: dict) -> list[ToolResolution]:
    """Match each slicer tool against bound spools using exact (type, color_hex).

    slots_response is the JSON from GET /api/slots:
      {"aces": [{"index": N, "slots": [{"slot": S, "spool": {...} | null}]}]}

    Returns one ToolResolution per tool (same order as input).
    """
    # Build flat list of (ace, slot, material_lower, color_lower, spool_id, name)
    all_bindings: list[tuple[int, int, str, str, int, str]] = []
    for ace_block in (slots_response.get("aces") or []):
        ace_idx = int(ace_block["index"])
        for slot_block in (ace_block.get("slots") or []):
            spool = slot_block.get("spool")
            if not spool:
                continue
            mat = (spool.get("material") or "").strip().upper()
            col = (spool.get("color") or "").strip().lower().lstrip("#")
            all_bindings.append((
                ace_idx,
                int(slot_block["slot"]),
                mat,
                col,
                int(spool["spool_id"]),
                spool.get("name") or "",
            ))

    resolutions: list[ToolResolution] = []
    for tool in tools:
        tmeta = ToolMeta(
            index=len(resolutions),
            type=tool["type"].upper(),
            color=tool["color"].lower().lstrip("#"),
        )
        candidates = [
            Candidate(ace=ace, slot=slot, spool_id=sid, spool_name=name)
            for ace, slot, mat, col, sid, name in all_bindings
            if mat == tmeta.type and col == tmeta.color
        ]
        if len(candidates) == 1:
            mq = "exact"
            resolved = candidates[0]
        elif len(candidates) > 1:
            mq = "ambiguous"
            resolved = None
        else:
            mq = "none"
            resolved = None

        resolutions.append(ToolResolution(
            tool=tmeta,
            match_quality=mq,
            candidates=candidates,
            resolved=resolved,
        ))
    return resolutions
```

_LAYER_RE = re.compile(r"^\s*;\s*-+\s*layer\s+(\d+)\s*-+", re.IGNORECASE)
_TOOL_RE = re.compile(r"^\s*T(\d+)\s*(?:;.*)?$")


def plan_swaps(resolutions: list[ToolResolution], gcode_lines: list[str]) -> list[SwapEvent]:
    """Greedy swap planner: minimize number of physical-head reassignments.

    Algorithm:
      1. Assign the first 4 distinct resolved tools (in order of first use in
         gcode) to physical heads 0-3.
      2. Walk gcode line-by-line; when a Tn is encountered:
         - If the tool's filament is already on a head → use that head (no swap).
         - Else find the "best" head to evict (greedy: first head whose tool
           won't appear again in the remaining gcode). Assign new tool to that
           head → one swap event.
      3. Unresolved tools (match_quality != "exact" and no resolved) are skipped.
    """
    # Build tool_index → Candidate map (only resolved tools)
    resolved: dict[int, Candidate] = {}
    for r in resolutions:
        if r.resolved is not None:
            resolved[r.tool.index] = r.resolved

    # Collect order-of-appearance tool indices from gcode
    appearance_order: list[int] = []
    seen: set[int] = set()
    future_uses: dict[int, list[int]] = {}  # tool_idx → list of line numbers
    for lineno, line in enumerate(gcode_lines):
        m = _TOOL_RE.match(line)
        if m:
            t = int(m.group(1))
            if t not in seen:
                seen.add(t)
                if t in resolved:
                    appearance_order.append(t)
            future_uses.setdefault(t, []).append(lineno)

    # Initial head assignment: first 4 distinct resolved tools in appearance order
    head_to_tool: dict[int, Optional[int]] = {0: None, 1: None, 2: None, 3: None}
    tool_to_head: dict[int, int] = {}
    next_head = 0
    for t in appearance_order[:4]:
        if t not in tool_to_head:
            head_to_tool[next_head] = t
            tool_to_head[t] = next_head
            next_head += 1

    swaps: list[SwapEvent] = []
    current_layer = 0

    for lineno, line in enumerate(gcode_lines):
        lm = _LAYER_RE.match(line)
        if lm:
            current_layer = int(lm.group(1))
            continue

        tm = _TOOL_RE.match(line)
        if not tm:
            continue
        t = int(tm.group(1))
        if t not in resolved:
            continue  # unresolved or tool with no binding

        if t in tool_to_head:
            # Already assigned; assign the resolution's physical_head pointer
            resolutions[t].physical_head = tool_to_head[t]
            continue

        # Need to evict a head: pick the head whose tool has the latest next use
        # (greedy look-ahead). Among tied/exhausted tools, pick lowest head index.
        remaining_uses: dict[int, int] = {}
        for h, ht in head_to_tool.items():
            if ht is None:
                remaining_uses[h] = -1  # free slot; best candidate
                continue
            uses = [u for u in future_uses.get(ht, []) if u > lineno]
            remaining_uses[h] = min(uses) if uses else -1  # -1 = never used again

        # Prefer heads that are never-used-again (remaining_uses[h] == -1)
        evict_head = min(
            remaining_uses,
            key=lambda h: (remaining_uses[h] != -1, remaining_uses[h], h)
        )

        old_tool = head_to_tool[evict_head]
        old_cand = resolved[old_tool] if old_tool is not None and old_tool in resolved else None
        new_cand = resolved[t]

        swaps.append(SwapEvent(
            line=lineno,
            layer=current_layer,
            head=evict_head,
            from_ace=old_cand.ace if old_cand else 0,
            from_slot=old_cand.slot if old_cand else 0,
            to_ace=new_cand.ace,
            to_slot=new_cand.slot,
        ))

        # Update assignments
        if old_tool is not None and old_tool in tool_to_head:
            del tool_to_head[old_tool]
        head_to_tool[evict_head] = t
        tool_to_head[t] = evict_head
        resolutions[t].physical_head = evict_head

    # Set physical_head for initial assignments
    for h, t in head_to_tool.items():
        if t is not None and t < len(resolutions) and resolutions[t].physical_head is None:
            resolutions[t].physical_head = h

    return swaps
```

def rewrite_gcode(lines: list[str], resolutions: list[ToolResolution],
                  swaps: list[SwapEvent]) -> list[str]:
    """Rewrite gcode: remap logical T4-T7 to physical T0-T3; insert PARK/LOAD at swaps.

    Logical tool N maps to physical head resolutions[N].physical_head.
    Swap events are keyed by line number; we insert PARK+LOAD before that line.
    """
    # Build logical → physical mapping
    logical_to_physical: dict[int, int] = {}
    for r in resolutions:
        if r.physical_head is not None:
            logical_to_physical[r.tool.index] = r.physical_head

    # Build set of line numbers that need PARK/LOAD inserted before them
    swap_by_line: dict[int, SwapEvent] = {s.line: s for s in swaps}

    # Build header comment block
    header_comments = []
    for r in resolutions:
        cand = r.resolved
        if cand:
            header_comments.append(
                f"; multiace.tool{r.tool.index}: type={r.tool.type} "
                f"color=#{r.tool.color} head={r.physical_head} "
                f"source=ace{cand.ace}/slot{cand.slot} spool_id={cand.spool_id}"
            )
        else:
            header_comments.append(
                f"; multiace.tool{r.tool.index}: type={r.tool.type} "
                f"color=#{r.tool.color} head=UNRESOLVED"
            )
    total_swaps = len(swaps)
    swap_layers = ",".join(str(s.layer) for s in swaps) if swaps else "none"
    unresolved_count = sum(1 for r in resolutions if r.resolved is None)
    status = "ready" if unresolved_count == 0 else "pending"
    header_comments += [
        f"; multiace.swaps: {total_swaps} (at layers {swap_layers})",
        f"; multiace.status: {status}",
    ]

    # Find insertion point for header block (after first '; generated by' or at line 0)
    header_insert_at = 0
    for i, line in enumerate(lines):
        if line.strip().startswith("; generated by") or line.strip().startswith("; filament_type"):
            header_insert_at = i
            break

    out: list[str] = []
    for lineno, line in enumerate(lines):
        # Insert header block after the slicer header comments
        if lineno == header_insert_at + 1 and header_comments:
            out.extend(header_comments)
            header_comments = []  # only inject once

        # Insert PARK + LOAD before a swap boundary line
        if lineno in swap_by_line:
            sw = swap_by_line[lineno]
            out.append(f"ACE_PARK_HEAD HEAD={sw.head}")
            out.append(f"ACE_LOAD_HEAD HEAD={sw.head} ACE={sw.to_ace} SLOT={sw.to_slot}")

        # Remap logical T to physical T
        tm = _TOOL_RE.match(line)
        if tm:
            logical = int(tm.group(1))
            phys = logical_to_physical.get(logical, logical)
            out.append(f"T{phys}" + (line[tm.end():] if tm.end() < len(line) else ""))
        else:
            out.append(line)

    return out


def write_sidecar(gcode_path: Path, resolutions: list[ToolResolution],
                  swaps: list[SwapEvent], status: str,
                  reason: Optional[str] = None, errors: Optional[list] = None) -> None:
    """Write <gcode_path>.multiace.json atomically."""
    tools_dict: dict[str, dict] = {}
    for r in resolutions:
        cand = r.resolved
        tools_dict[str(r.tool.index)] = {
            "type": r.tool.type,
            "color": f"#{r.tool.color}",
            "match_quality": r.match_quality,
            "candidates": [
                {"ace": c.ace, "slot": c.slot, "spool_id": c.spool_id, "spool_name": c.spool_name}
                for c in r.candidates
            ],
            "resolved": (
                {"ace": cand.ace, "slot": cand.slot, "spool_id": cand.spool_id}
                if cand else None
            ),
            "physical_head": r.physical_head,
        }
    swaps_list = [

            "line": s.line,
            "layer": s.layer,
            "head": s.head,
            "from": {"ace": s.from_ace, "slot": s.from_slot},

        """Re-run match + plan against current slot bindings and rewrite the sidecar.

        Imports the postprocessor at call time to avoid circular imports and to
        allow tools/ to remain a standalone script.
        """
        import sys as _sys
        tools_dir = Path(__file__).resolve().parents[4] / "tools"
        if str(tools_dir) not in _sys.path:
            _sys.path.insert(0, str(tools_dir))
        import importlib
        pp = importlib.import_module("multiace_postprocess")

        lines = gcode_path.read_text(encoding="utf-8", errors="replace").splitlines()
        tools = pp.parse_header(lines)
        if tools is None:
            return {"status": "error", "reason": "no_8_tool_header", "tools": {}, "swaps": []}
        resolutions = pp.match_tools(tools, slots_resp)
        swaps = pp.plan_swaps(resolutions, lines)
        new_lines = pp.rewrite_gcode(lines, resolutions, swaps)

        unresolved = [r for r in resolutions if r.match_quality != "exact" and r.resolved is None]
        ambiguous = [r for r in resolutions if r.match_quality == "ambiguous"]
        if ambiguous:
            status, reason = "pending", "ambiguous_match"
        elif unresolved:
            status, reason = "pending", "missing_bindings"
        else:
            status, reason = "ready", None

        gcode_path.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
        pp.write_sidecar(gcode_path, resolutions, swaps, status, reason)
        return {"status": status, "reason": reason,
                "tools": {str(r.tool.index): {
                    "type": r.tool.type, "color": f"#{r.tool.color}",

                    "physical_head": r.physical_head,
                } for r in resolutions},

        toast("Smart-swap not available. Load the spool manually via FilamentHub picker.", "error");
      }
    });
  });
}

            if rev_btn:
                print("OK: Re-validate button present")
            else:

    @app.post("/api/print_queue/{gcode_filename:path}/print")
    async def start_print_queue_file(request: Request, gcode_filename: str) -> dict:
        """Issue SDCARD_PRINT_FILE for the specified gcode file via Moonraker."""
        # Validate filename to prevent injection (no path traversal)
        if "/" in gcode_filename or ".." in gcode_filename:
            raise HTTPException(400, "invalid filename")
        try:

                f"SDCARD_PRINT_FILE FILENAME={gcode_filename}"
            )
        except MoonrakerError as e:
            raise HTTPException(502, str(e))
        return {"ok": True, "filename": gcode_filename}
